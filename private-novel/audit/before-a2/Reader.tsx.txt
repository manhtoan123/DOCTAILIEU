import {useCallback,useEffect,useMemo,useRef,useState,type CSSProperties} from 'react';
import {useLiveQuery} from 'dexie-react-hooks';
import {db} from '../db';
import {defaults,paragraphRanges,parseEntry,uid,type ReaderSettings,type Chapter} from '../domain';
import {convert,lookup} from '../language-client';
import type {Token} from '../language';
import {TikTokPlayer,tiktokVoices,voiceOptions,sentences,type SpeechState,type Sentence,type TikTokVoice,TtsProvider} from '../tts';
import {download} from '../backup';
import {saveBookNames} from '../book-names';
import {Modal,errorText} from './common';
import {DisplaySettings,ReaderPanel,ReaderHelp,readerFonts} from './ReaderTools';
export default function Reader({bookId,onBack,notify}:{bookId:string;onBack:()=>void;notify:(s:string)=>void}){
  const book=useLiveQuery(()=>db.books.get(bookId),[bookId]);
  const list=useLiveQuery(async()=>{const items:Pick<Chapter,'id'|'title'|'order'>[]=[];await db.chapters.where('bookId').equals(bookId).each(c=>{items.push({id:c.id,title:c.title,order:c.order});});return items.sort((a,b)=>a.order-b.order);},[bookId]);
  const [chapterId,setChapterId]=useState('');const chapter=useLiveQuery(()=>chapterId?db.chapters.get(chapterId):undefined,[chapterId]);
  const names=useLiveQuery(()=>db.names.where('bookId').equals(bookId).toArray(),[bookId]);
  const [settings,setSettings]=useState<ReaderSettings>(defaults);const[tool,setTool]=useState<''|'settings'|'speech'|'help'>('');const[focusMode,setFocusMode]=useState(false);const[wordEditing,setWordEditing]=useState(false);const[toc,setToc]=useState(false);const[tocFilter,setTocFilter]=useState('');const[tocPage,setTocPage]=useState(0);
  const[mode,setMode]=useState<'raw'|'convert'>('raw');const[tokens,setTokens]=useState<Token[]>([]);const[converting,setConverting]=useState(false);const[tempNames,setTempNames]=useState<{key:string;value:string}[]>([]);const[ttsTextOverride,setTtsTextOverride]=useState('');const[ttsEditor,setTtsEditor]=useState(false);const[ttsDraft,setTtsDraft]=useState<string[]>([]);
  const[edit,setEdit]=useState<string|null>(null);const editRef=useRef<HTMLTextAreaElement>(null);const[selected,setSelected]=useState<Token>();const[nameKey,setNameKey]=useState('');const[nameValue,setNameValue]=useState('');const[meaning,setMeaning]=useState({meanings:[] as string[],hanviet:''});const[namePanel,setNamePanel]=useState(false);
  const[autoNext,setAutoNext]=useState(true);const[speech,setSpeech]=useState<SpeechState>('idle');const[sentence,setSentence]=useState<Sentence>();const[nameDraft,setNameDraft]=useState('');const[nameSaving,setNameSaving]=useState(false);const[remoteConsent,setRemoteConsent]=useState(false);
  const nameDirty=useRef(false);const nameDraftRevision=useRef(0);
  const conversionSource=useRef<{id:string;revision:number;names:typeof names;tempNames:typeof tempNames}|undefined>(undefined);
  const endedRef=useRef<()=>void>(()=>{});const continueRef=useRef(false);const playerRef=useRef<TikTokPlayer|null>(null);
  if(!playerRef.current)playerRef.current=new TikTokPlayer((state,s,msg)=>{setSpeech(state);setSentence(s);if(msg)notify(msg);},()=>endedRef.current());
  const progressTimer=useRef<ReturnType<typeof setTimeout>|undefined>(undefined);const position=useRef({paragraph:0,offset:0});const article=useRef<HTMLElement>(null);
  const save=useCallback(()=>{if(chapter)void db.progress.put({bookId,chapterId:chapter.id,paragraph:position.current.paragraph,offset:position.current.offset,revision:chapter.revision,updatedAt:Date.now()}).catch(e=>notify(errorText(e)));},[bookId,chapter,notify]);
  const settingsLoaded=useRef(false);
  useEffect(()=>{void db.settings.get('reader').then(row=>{if(row)setSettings({...defaults,...row.value as ReaderSettings});settingsLoaded.current=true;});},[]);
  useEffect(()=>{if(settingsLoaded.current)void db.settings.put({key:'reader',value:settings});},[settings]);
  useEffect(()=>{let alive=true;void db.progress.get(bookId).then(p=>{if(alive&&list?.length)setChapterId(old=>list.some(c=>c.id===old)?old:list.some(c=>c.id===p?.chapterId)?p!.chapterId:list[0].id);});return()=>{alive=false;};},[bookId,list]);
  useEffect(()=>()=>{stopReading();},[]);
  useEffect(()=>{let alive=true;conversionSource.current=undefined;setTokens([]);if(mode!=='convert'||!chapter){setConverting(false);return;}setConverting(true);void convert(chapter.text,[...(names||[]),...tempNames]).then(t=>{if(alive){conversionSource.current={id:chapter.id,revision:chapter.revision,names,tempNames};setTokens(t);}}).catch(e=>{if(alive){notify(errorText(e));setMode('raw');}}).finally(()=>{if(alive)setConverting(false);});return()=>{alive=false;};},[chapter?.id,chapter?.revision,mode,names,tempNames]);
  const conversionReady=conversionSource.current?.id===chapter?.id&&conversionSource.current?.revision===chapter?.revision&&conversionSource.current?.names===names&&conversionSource.current?.tempNames===tempNames;
  const waitingForConvert=mode==='convert'&&(!conversionReady||converting);
  const renderedText=mode==='convert'?(conversionReady?tokens.map(t=>t.targetText).join(''):''):chapter?.text||'';
  const speechText=ttsTextOverride||renderedText;
  const paragraphs=useMemo(()=>paragraphRanges(renderedText),[renderedText]);
  const blocks=useMemo(()=>{
    if(mode==='raw')return [];
    const rows:{token:Token;text:string}[][]=[[]];for(const token of tokens){const pieces=token.targetText.split('\n');pieces.forEach((text,i)=>{if(text)rows[rows.length-1].push({token,text});if(i<pieces.length-1)rows.push([]);});}return rows;
  },[tokens,mode]);
  useEffect(()=>{if(!chapter||converting||!paragraphs.length)return;let alive=true;let ready=false;let timeout:ReturnType<typeof setTimeout>;let observer:IntersectionObserver|undefined;
    void db.progress.get(bookId).then(p=>{if(!alive)return;const target=p?.chapterId===chapter.id?Math.min(p.paragraph,paragraphs.length-1):0;position.current={paragraph:target,offset:0};timeout=setTimeout(()=>{if(!alive)return;article.current?.querySelector(`[data-paragraph="${target}"]`)?.scrollIntoView({block:'start'});ready=true;},80);
      observer=new IntersectionObserver(entries=>{if(!ready)return;const visible=entries.filter(e=>e.isIntersecting).sort((a,b)=>a.boundingClientRect.top-b.boundingClientRect.top);if(visible[0]){position.current.paragraph=Number((visible[0].target as HTMLElement).dataset.paragraph);clearTimeout(progressTimer.current);progressTimer.current=setTimeout(save,600);}}, {rootMargin:'-15% 0px -60% 0px'});
      article.current?.querySelectorAll('[data-paragraph]').forEach(p=>observer!.observe(p));
    });const visibility=()=>{if(ready)save();};document.addEventListener('visibilitychange',visibility);return()=>{alive=false;clearTimeout(timeout);clearTimeout(progressTimer.current);observer?.disconnect();document.removeEventListener('visibilitychange',visibility);if(ready)save();};
  },[chapter?.id,chapter?.revision,converting,renderedText]);
  const currentIndex=list?.findIndex(c=>c.id===chapterId)??0;
  function stopReading(){continueRef.current=false;playerRef.current?.stop();}
  function changeChapter(id:string,keepPlaying=false){save();stopReading();continueRef.current=keepPlaying;setTtsTextOverride('');setChapterId(id);setToc(false);setSelected(undefined);}
  function openNames(){setTool('');if(!nameDirty.current)setNameDraft((names||[]).map(n=>`${n.key}=${n.value}`).join('\n'));setNamePanel(true);}
  function toggleTool(next:typeof tool){setNamePanel(false);setTool(old=>old===next?'':next);}
  useEffect(()=>{const escape=(e:KeyboardEvent)=>{if(e.key==='Escape'&&!tool&&!namePanel&&!toc&&!selected&&edit===null&&!ttsEditor)setFocusMode(false);};document.addEventListener('keydown',escape);return()=>document.removeEventListener('keydown',escape);},[tool,namePanel,toc,selected,edit,ttsEditor]);
  async function applyNames(run=false){if(nameSaving)return;setNameSaving(true);const draftRevision=nameDraftRevision.current;try{const count=await saveBookNames(bookId,nameDraft);if(draftRevision===nameDraftRevision.current)nameDirty.current=false;stopReading();if(run){setMode('convert');setNamePanel(false);}notify(`Đã cập nhật ${count} dòng Name cho truyện này.`);}catch(e){notify(errorText(e));}finally{setNameSaving(false);}}
  function openTtsEditor(){setTool('');setTtsDraft(sentences(speechText).map(item=>item.text));setTtsEditor(true);}
  function saveTtsDraft(){const text=ttsDraft.map(item=>item.trim()).filter(Boolean).join('\n');setTtsTextOverride(text);setTtsEditor(false);stopReading();notify('Đã áp dụng văn bản nghe cho phiên này.');}
  function play(){if(waitingForConvert||!speechText.trim())return;if(!remoteConsent){setTool('speech');notify('Tích đồng ý gửi câu đang đọc đến gateway trong bảng Nghe sách trước khi phát.');return;}const option=voiceOptions.find(v=>v.id===settings.voiceId);if(!option?.available||option.provider!=='tiktok'){notify('Gateway của giọng này chưa được cấu hình; hiện TikTok TTS là giọng được hỗ trợ.');return;}playerRef.current!.start(speechText,option as TikTokVoice,{rate:settings.ttsRate,pitch:settings.ttsPitch,flex:settings.ttsFlex});}
  endedRef.current=()=>{if(autoNext&&list?.[currentIndex+1])changeChapter(list[currentIndex+1].id,true);};
  useEffect(()=>{setTtsTextOverride('');},[chapter?.id,mode]);
  useEffect(()=>{if(continueRef.current&&autoNext&&chapter?.id===chapterId&&!waitingForConvert&&speechText&&remoteConsent){continueRef.current=false;const option=voiceOptions.find(v=>v.id===settings.voiceId)||tiktokVoices[0];if(option.provider==='tiktok')playerRef.current!.start(speechText,option as TikTokVoice,{rate:settings.ttsRate,pitch:settings.ttsPitch,flex:settings.ttsFlex});}},[chapterId,chapter?.id,speechText,waitingForConvert,remoteConsent,autoNext,settings.voiceId,settings.ttsRate,settings.ttsPitch,settings.ttsFlex]);
  async function selectToken(token:Token){setSelected(token);setNameKey(token.source);setNameValue(token.targetText.trim());try{setMeaning(await lookup(token.source));}catch(e){notify(errorText(e));}}
  async function saveName(permanent:boolean){if(!nameKey.trim()||!nameValue.trim())return;if(nameKey.length>256||nameValue.length>4096){notify('Tên hoặc nghĩa quá dài.');return;}if(permanent){const old=await db.names.where('[bookId+key]').equals([bookId,nameKey.trim()]).first();await db.names.put({id:old?.id||uid(),bookId,key:nameKey.trim(),value:nameValue.trim()});}else setTempNames(items=>[...items.filter(n=>n.key!==nameKey),{key:nameKey.trim(),value:nameValue.trim()}]);setSelected(undefined);notify(permanent?'Đã lưu Name cho truyện này.':'Đã sửa tạm trong phiên đọc.');}
  async function saveEdit(splitAt?:number){if(!chapter||edit===null)return;if(edit.length>128000){notify('Mỗi phần tối đa 128.000 ký tự. Hãy chia thành nhiều phần.');return;}const delta=edit.length-chapter.text.length;
    await db.transaction('rw',[db.chapters,db.books],async()=>{if(splitAt&&splitAt>0&&splitAt<edit.length){const next=list?.[currentIndex+1];const order=next?(chapter.order+next.order)/2:chapter.order+1;await db.chapters.add({...chapter,id:uid(),order,title:chapter.title+' · tiếp',text:edit.slice(splitAt),revision:1});await db.chapters.update(chapter.id,{text:edit.slice(0,splitAt),revision:chapter.revision+1});await db.books.update(bookId,{chapterCount:(book?.chapterCount||0)+1,charCount:(book?.charCount||0)+delta,updatedAt:Date.now()});}else{await db.chapters.update(chapter.id,{text:edit,revision:chapter.revision+1});await db.books.update(bookId,{charCount:(book?.charCount||0)+delta,updatedAt:Date.now()});}});setEdit(null);stopReading();}
  const filtered=list?.filter(c=>c.title.toLocaleLowerCase().includes(tocFilter.toLocaleLowerCase()))||[];
  return <div className={`reader theme-${settings.theme} ${focusMode?'reader-focus':''}`} style={{'--tts-highlight':settings.highlightColor,...(settings.background?{'--paper':settings.background}:{}),...(settings.textColor?{'--ink':settings.textColor}:{})} as CSSProperties}>
    <header className="reader-header"><button onClick={()=>{save();onBack();}}>← Tủ sách</button><span>Thư phòng / Đọc truyện</span><button aria-pressed={focusMode} onClick={()=>{setFocusMode(true);setTool('');setNamePanel(false);}}>Chỉ đọc</button></header>
    <div className="reader-book-heading"><h1>{book?.title}</h1><span>{chapter?.title||'Đang mở…'}</span></div>
    <nav className="chapter-navigation chapter-navigation-top" aria-label="Điều hướng chương đầu trang"><button disabled={currentIndex<=0} onClick={()=>changeChapter(list![currentIndex-1].id)}>← Chương trước</button><button onClick={()=>{setTool('');setNamePanel(false);setToc(true);}}>☷ Mục lục</button><button disabled={!list?.[currentIndex+1]} onClick={()=>changeChapter(list![currentIndex+1].id)}>Chương sau →</button></nav>
    <div className="reader-actions"><div className="segmented"><button aria-pressed={mode==='raw'} className={mode==='raw'?'active':''} onClick={()=>{stopReading();setTtsTextOverride('');setMode('raw');}}>Bản gốc</button><button aria-pressed={mode==='convert'} className={mode==='convert'?'active':''} onClick={()=>{stopReading();setTtsTextOverride('');setMode('convert');}}>Convert</button></div><button aria-pressed={wordEditing} disabled={mode!=='convert'} onClick={()=>setWordEditing(v=>!v)}>Sửa từ: {wordEditing?'bật':'tắt'}</button><button aria-expanded={namePanel} onClick={()=>namePanel?setNamePanel(false):openNames()}>Name ({names?.length||0})</button><button aria-expanded={tool==='speech'} onClick={()=>toggleTool('speech')}>▶ Nghe sách</button><details className="reader-more"><summary>Công cụ ▾</summary><div><button onClick={()=>{setTool('');setNamePanel(false);setEdit(chapter?.text||'');}}>Sửa chương</button><button onClick={()=>{if(chapter)download(new Blob([chapter.text],{type:'text/plain;charset=utf-8'}),`${chapter.title.replace(/[<>:"/\\|?*]/g,'_')}.txt`);}}>Xuất TXT</button><button onClick={()=>toggleTool('help')}>Hướng dẫn thao tác</button></div></details></div>
    {mode==='convert'&&wordEditing&&<p className="word-mode-hint">Chạm vào từ để chọn nghĩa hoặc sửa Name. Tắt “Sửa từ” để đọc bình thường.</p>}
    <article ref={article} className="prose" style={{maxWidth:settings.width,fontSize:settings.fontSize,lineHeight:settings.lineHeight,fontFamily:readerFonts[settings.font]||readerFonts.arial,textAlign:settings.align||'left',textShadow:settings.textShadow?'0 0 1px currentColor':'none'}}>
      <div className="chapter-kicker">CHƯƠNG / PHẦN {Math.max(0,currentIndex)+1} TRÊN {list?.length||0}</div>
      {waitingForConvert?<p className="notice">Đang Convert trong Worker…</p>:paragraphs.map((p,i)=><p key={`${chapterId}-${mode}-${i}`} data-paragraph={i} data-empty={!p.text.trim()} className={!ttsTextOverride&&sentence&&sentence.start<p.start+p.text.length&&sentence.end>p.start?'speaking':''}>{mode==='convert'?(blocks[i]||[]).map((x,j)=>wordEditing?<button type="button" key={j} className="word-token" onClick={()=>{setTool('');setNamePanel(false);void selectToken(x.token);}}>{x.text}</button>:<span key={j}>{x.text}</span>):p.text}{!p.text.trim()&&'\u00a0'}</p>)}
      <nav className="chapter-navigation" aria-label="Điều hướng chương cuối trang"><button disabled={currentIndex<=0} onClick={()=>changeChapter(list![currentIndex-1].id)}>← Chương trước</button><button onClick={()=>{setTool('');setNamePanel(false);setToc(true);}}>☷ Mục lục</button><button disabled={!list?.[currentIndex+1]} onClick={()=>changeChapter(list![currentIndex+1].id)}>Chương sau →</button></nav>
    </article>
    {tool==='speech'&&<ReaderPanel title="Nghe sách" kind="speech" onClose={()=>setTool('')}><p className="muted">TikTok cần gateway hoạt động. Câu đang đọc sẽ được gửi đến gateway. Các giọng ghi “chưa cấu hình” chưa phát được.</p><label className="check"><input type="checkbox" checked={remoteConsent} onChange={e=>{setRemoteConsent(e.target.checked);if(!e.target.checked)stopReading();}}/>Đồng ý gửi câu đang đọc đến gateway TTS</label><div className="tts-bar"><button className="primary" disabled={waitingForConvert||!chapter||!remoteConsent||speech==='loading'} onClick={()=>speech==='playing'?playerRef.current?.pause():speech==='paused'?playerRef.current?.resume():play()}>{speech==='playing'?'Ⅱ Tạm dừng':speech==='paused'?'▶ Tiếp tục':speech==='loading'?'Đang tải TikTok…':'▶ Nghe TikTok'}</button><button aria-label="Dừng đọc" onClick={()=>stopReading()}>■</button><button onClick={openTtsEditor}>Câu ({sentences(speechText).length})</button><label>Giọng<select aria-label="Giọng đọc" value={settings.voiceId} onChange={e=>{stopReading();setSettings(s=>({...s,voiceId:e.target.value}));}}>{(['bing','google-cloud','google-translate','tiktok'] as TtsProvider[]).map(provider=><optgroup key={provider} label={provider==='bing'?'Dịch vụ giọng nói Bing':provider==='google-cloud'?'Dịch vụ giọng nói Google Cloud':provider==='google-translate'?'Giọng Chị Google':'Giọng đọc của TikTok'}>{voiceOptions.filter(v=>v.provider===provider).map(v=><option key={v.id} value={v.id} disabled={!v.available}>{v.name}{v.available?'':' · chưa cấu hình'}</option>)}</optgroup>)}</select></label><label>Tốc độ<input aria-label="Tốc độ đọc" type="range" min="0.5" max="5" step="0.1" value={settings.ttsRate} onChange={e=>{stopReading();setSettings(s=>({...s,ttsRate:+e.target.value}));}}/></label><output>{settings.ttsRate}</output><label>Cao độ<input aria-label="Độ cao giọng" type="range" min="0.5" max="2" step="0.1" value={settings.ttsPitch} onChange={e=>setSettings(s=>({...s,ttsPitch:+e.target.value}))}/></label><output>{settings.ttsPitch}</output><label>Ngữ cảnh<input aria-label="Điều chỉnh theo ngữ cảnh" type="range" min="0" max="2" step="0.1" value={settings.ttsFlex} onChange={e=>setSettings(s=>({...s,ttsFlex:+e.target.value}))}/></label><output>{settings.ttsFlex}</output><label>Màu highlight<input aria-label="Màu làm nổi bật văn bản" type="color" value={settings.highlightColor} onChange={e=>setSettings(s=>({...s,highlightColor:e.target.value}))}/></label><button onClick={()=>void db.settings.put({key:'reader',value:settings})}>Lưu cài đặt</button><label className="check"><input type="checkbox" checked={autoNext} onChange={e=>{setAutoNext(e.target.checked);if(!e.target.checked)continueRef.current=false;}}/>Qua chương</label><small>TikTok TTS · tiếng Việt</small></div><p className="muted">Cao độ và ngữ cảnh chỉ có tác dụng nếu gateway hỗ trợ. Khi sửa câu đọc, highlight trên bản gốc tạm tắt để tránh tô sai đoạn.</p></ReaderPanel>}
    {toc&&<Modal title="Mục lục" onClose={()=>setToc(false)}><input aria-label="Tìm chương" placeholder="Tìm chương…" value={tocFilter} onChange={e=>{setTocFilter(e.target.value);setTocPage(0);}}/><div className="toc">{filtered.slice(tocPage*60,tocPage*60+60).map(c=><button className={c.id===chapterId?'current':''} key={c.id} onClick={()=>changeChapter(c.id)}>{c.title}</button>)}</div><div className="pager"><button disabled={!tocPage} onClick={()=>setTocPage(p=>p-1)}>Trước</button><span>{tocPage+1}</span><button disabled={(tocPage+1)*60>=filtered.length} onClick={()=>setTocPage(p=>p+1)}>Tiếp</button></div></Modal>}
    {edit!==null&&<Modal title="Sửa nội dung chương" onClose={()=>setEdit(null)} wide><p className="muted">Sửa bản lưu trong ứng dụng. File nguồn giữ nguyên. Đặt con trỏ để tách chương tại vị trí đó.</p><textarea ref={editRef} aria-label="Nội dung chương" rows={18} value={edit} onChange={e=>setEdit(e.target.value)}/><footer><button onClick={()=>void saveEdit(editRef.current?.selectionStart)}>Tách tại con trỏ</button><button className="primary" onClick={()=>void saveEdit()}>Lưu thay đổi</button></footer></Modal>}
    {selected&&<Modal title="Chỉnh Name / nghĩa" onClose={()=>setSelected(undefined)}><label>Chữ gốc<input value={nameKey} onChange={e=>setNameKey(e.target.value)}/></label><div className="pager"><button disabled={!chapter||selected.sourceStart<=0} onClick={()=>{if(chapter)void selectToken({...selected,sourceStart:Math.max(0,selected.sourceStart-1),source:chapter.text.slice(Math.max(0,selected.sourceStart-1),selected.sourceEnd)});}}>← Mở rộng trái</button><button disabled={!chapter||selected.sourceEnd>=chapter.text.length} onClick={()=>{if(chapter)void selectToken({...selected,sourceEnd:selected.sourceEnd+1,source:chapter.text.slice(selected.sourceStart,selected.sourceEnd+1)});}}>Mở rộng phải →</button></div><p>Hán-Việt: {meaning.hanviet||'Chưa có phiên âm'}</p><div className="choices">{meaning.meanings.slice(0,100).map((m,i)=><button key={i} onClick={()=>setNameValue(m)}>{m}</button>)}</div><label>Nghĩa / tên Việt<input value={nameValue} onChange={e=>setNameValue(e.target.value)}/></label><div className="choices"><button onClick={()=>setNameValue(v=>v.toLocaleUpperCase('vi'))}>CHỮ HOA</button><button onClick={()=>setNameValue(v=>v.toLocaleLowerCase('vi'))}>chữ thường</button></div><footer><button onClick={()=>void saveName(false)}>Sửa tạm</button><button className="primary" onClick={()=>void saveName(true)}>Lưu Name cho truyện</button></footer></Modal>}
    {namePanel&&<ReaderPanel title="Name riêng của truyện" kind="names" onClose={()=>setNamePanel(false)}>
      <div className="name-toolbar"><button disabled={nameSaving} onClick={()=>void applyNames()}>Lưu lại</button><button disabled={nameSaving} onClick={()=>void applyNames(true)}>▶ Chạy</button><button onClick={()=>download(new Blob([nameDraft]),'NameList.txt')}>Xuất dữ liệu</button></div>
      <p className="muted">Mỗi dòng: chữ Trung=tên Việt. Lưu cập nhật tên trùng và thêm tên mới. Xóa dòng trong ô không xóa tên đã lưu; dùng nút × bên dưới.</p>
      <textarea aria-label="Nội dung Name List" disabled={nameSaving} rows={10} value={nameDraft} onChange={e=>{nameDirty.current=true;nameDraftRevision.current++;setNameDraft(e.target.value);}} placeholder="张三=Trương Tam"/>
      <label>Nhập dữ liệu Name<input type="file" accept=".txt" aria-label="Nhập Name List" disabled={nameSaving} onChange={async e=>{const f=e.target.files?.[0];if(!f)return;try{if(f.size>2*1048576)throw new Error('Name List tối đa 2 MB.');const text=await f.text();nameDirty.current=true;nameDraftRevision.current++;setNameDraft(old=>old+(old?'\n':'')+text);notify('Đã đưa file vào ô soạn. Bấm Lưu lại hoặc Chạy để áp dụng.');}catch(err){notify(errorText(err));}}}/></label>
      <h3>Đã lưu ({names?.length||0})</h3><div className="name-list">{names?.slice(0,200).map(n=><div key={n.id}><span>{n.key} = {n.value}</span><button aria-label={`Xóa Name ${n.key}`} disabled={nameSaving} onClick={async()=>{try{await db.names.delete(n.id);setNameDraft(old=>old.split('\n').filter(line=>parseEntry(line)?.[0]!==n.key).join('\n'));}catch(e){notify(errorText(e));}}}>×</button></div>)}</div>{(names?.length||0)>200&&<p className="muted">Hiển thị 200 mục đầu. Ô soạn và file xuất chứa toàn bộ Name.</p>}
    </ReaderPanel>}
    {tool==='settings'&&<ReaderPanel title="Cài đặt hiển thị" onClose={()=>setTool('')}><DisplaySettings settings={settings} setSettings={setSettings} onNames={openNames} onSpeech={()=>setTool('speech')} onHelp={()=>setTool('help')}/><button className="focus-toggle" aria-pressed={focusMode} onClick={()=>{setFocusMode(v=>!v);setTool('');}}>{focusMode?'Hiện thanh công cụ':'Chế độ chỉ đọc'}</button></ReaderPanel>}
    {tool==='help'&&<ReaderPanel title="Hướng dẫn thao tác" kind="help" onClose={()=>setTool('')}><ReaderHelp/></ReaderPanel>}
    {ttsEditor&&<Modal title="Sửa câu đọc trong phiên" onClose={()=>setTtsEditor(false)} wide><p className="muted">Chỉ thay nội dung gửi đến TTS trong chương hiện tại. Bản truyện đã lưu không thay đổi.</p><div className="sentence-editor">{ttsDraft.map((text,i)=><label key={i}>Câu {i+1}<textarea rows={2} value={text} onChange={e=>setTtsDraft(old=>old.map((value,index)=>index===i?e.target.value:value))}/></label>)}</div><footer><button onClick={()=>{setTtsTextOverride('');setTtsEditor(false);stopReading();}}>Dùng bản truyện</button><button onClick={saveTtsDraft}>Áp dụng cho phiên</button></footer></Modal>}
    <div className="floating-tools">{focusMode&&<><button aria-label="Chương trước nhanh" disabled={currentIndex<=0} onClick={()=>changeChapter(list![currentIndex-1].id)}>‹</button><button aria-label="Chương sau nhanh" disabled={!list?.[currentIndex+1]} onClick={()=>changeChapter(list![currentIndex+1].id)}>›</button></>}{(speech==='playing'||speech==='paused'||speech==='loading')&&<button aria-label="Dừng đọc nhanh" onClick={()=>stopReading()}>■</button>}<button className="settings-trigger" aria-label="Cài đặt đọc truyện" aria-expanded={tool==='settings'} aria-controls="reader-settings" onClick={()=>toggleTool('settings')}>⚙</button></div>
  </div>;
}
