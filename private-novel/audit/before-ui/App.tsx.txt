import {useCallback,useEffect,useState} from 'react';
import {useLiveQuery} from 'dexie-react-hooks';
import {db,deleteBook} from '../db';
import {uid,formatBytes,type Book} from '../domain';
import {setLanguageStatus} from '../language-client';
import ImportDialog from './ImportDialog';
import DictionaryPanel from './DictionaryPanel';
import BackupPanel from './BackupPanel';
import Reader from './Reader';
import {Empty,Modal,errorText} from './common';
const sample=`Chương 1: Một chỗ ngồi bên cửa sổ\n\nBuổi sáng, nắng ghé qua khung cửa gỗ. Trên bàn có một cuốn sách chưa đọc hết và tách trà vừa pha.\n\nBạn mở trang sách, chỉnh cỡ chữ vừa mắt, rồi để câu chuyện dẫn mình đi. Không có thông báo, không có vội vàng. Chỉ có những trang chữ và một chút thời gian dành riêng cho mình.\n\nChương 2: Lối mòn trong vườn\n\nSau cơn mưa, khu vườn trở nên yên tĩnh. Tiếng lá xào xạc như một lời nhắc: đôi khi điều cần làm chỉ là chậm lại.\n\nBạn có thể thử nút Nghe truyện, đổi nền Sepia, hoặc quay về tủ sách để nhập TXT của mình.\n\nChương 3: Trang sách tiếp theo\n\nThư phòng ghi nhớ nơi bạn dừng lại. Khi trở về, câu chuyện vẫn chờ ở đó.\n\nĐây là văn bản mẫu tự viết để kiểm tra trình đọc và giọng nói.\n`;
export default function App(){
  const books=useLiveQuery(()=>db.books.where('status').equals('ready').reverse().sortBy('updatedAt'),[]);const staged=useLiveQuery(()=>db.books.where('status').equals('staging').toArray(),[]);
  const[tab,setTab]=useState<'library'|'dictionary'|'backup'|'about'>('library');const[active,setActive]=useState('');const[importing,setImporting]=useState(false);const[search,setSearch]=useState('');const[toast,setToast]=useState('');const[languageStatus,setStatus]=useState('');const[edit,setEdit]=useState<Book>();const[storage,setStorage]=useState<StorageEstimate>();
  const notify=useCallback((message:string)=>setToast(message),[]);
  useEffect(()=>{setLanguageStatus(setStatus);void navigator.storage?.estimate().then(setStorage);},[]);
  useEffect(()=>{if(!toast)return;const timer=setTimeout(()=>setToast(''),9000);return()=>clearTimeout(timer);},[toast]);
  async function demo(){try{const id=uid();const textParts=sample.split(/(?=Chương \d+:)/).filter(Boolean);await db.transaction('rw',[db.books,db.chapters],async()=>{await db.books.add({id,title:'Một khoảng bình yên',author:'Thư Phòng · truyện mẫu',status:'ready',chapterCount:textParts.length,charCount:sample.length,createdAt:Date.now(),updatedAt:Date.now()});await db.chapters.bulkAdd(textParts.map((text,order)=>({id:uid(),bookId:id,order,title:text.split('\n')[0],text,revision:1,sourceId:''})));});setActive(id);}catch(e){notify(errorText(e));}}
  const filtered=books?.filter(b=>`${b.title} ${b.author}`.toLocaleLowerCase().includes(search.toLocaleLowerCase()))||[];
  return <>{active?<Reader key={active} bookId={active} onBack={()=>setActive('')} notify={notify}/>:<div className="shell">
    <aside className="sidebar"><a className="brand" href="#" onClick={e=>{e.preventDefault();setTab('library');}}><svg viewBox="0 0 40 40" aria-hidden="true"><path d="M5 9c7-3 12-2 15 1 3-3 8-4 15-1v23c-7-3-12-2-15 1-3-3-8-4-15-1z"/><path d="M20 10v23"/></svg><span>thư phòng<small>MỘT GÓC RIÊNG ĐỂ ĐỌC</small></span></a>
    <nav aria-label="Điều hướng chính">{([['library','▤','Tủ sách'],['dictionary','文','Từ điển'],['backup','↗','Sao lưu'],['about','◷','Về thư phòng']] as const).map(([key,icon,label])=><button key={key} className={tab===key?'active':''} onClick={()=>setTab(key)}><span>{icon}</span>{label}{key==='library'&&<small>{books?.length||0}</small>}</button>)}</nav>
    <div className="sidebar-bottom"><span className="status-dot"/> Lưu trên thiết bị<small>Nội dung riêng, nhịp đọc riêng.</small><p>{formatBytes(storage?.usage||0)} đang dùng</p></div></aside>
    <main className="workspace"><header className="topbar"><span>THƯ VIỆN CÁ NHÂN</span><span className="private-badge">● Local-first</span></header>
    {tab==='library'&&<><section className="hero"><div><span className="eyebrow">CHÀO MỪNG BẠN TRỞ LẠI</span><h1>Hôm nay,<br/>đọc một chút nhé.</h1><p>Một chỗ yên tĩnh cho những câu chuyện của bạn.<br/>Đọc, nghe và tiếp tục từ trang đang dở.</p><button className="primary" onClick={()=>setImporting(true)}>＋ Thêm truyện TXT</button></div><div className="book-art" aria-hidden="true"><div className="art-book art-back"/><div className="art-book art-front"><span>NHỮNG<br/>TRANG<br/><em>chưa đọc</em></span><div className="art-line"/><small>THƯ PHÒNG</small></div><span className="art-orbit"/></div></section>
    <section className="library-section"><div className="section-heading"><div><h2>Tủ sách của bạn <span className="count">{books?.length||0}</span></h2><p className="muted">Những câu chuyện, luôn ở đây.</p></div><input className="search" aria-label="Tìm truyện" value={search} onChange={e=>setSearch(e.target.value)} placeholder="Tìm tên truyện, tác giả…"/></div>
      {!books?.length?<div className="empty-shelf"><Empty title="Trang đầu tiên đang chờ">Thêm một file TXT để bắt đầu. Bạn cũng có thể mở truyện mẫu để thử cỡ chữ và giọng đọc.</Empty><div className="center-actions"><button className="primary" onClick={()=>setImporting(true)}>Chọn file TXT</button><button onClick={()=>void demo()}>Đọc truyện mẫu →</button></div></div>:<div className="book-grid">{filtered.map((b,i)=><article className="book-card" key={b.id}><button className={`cover cover-${i%4}`} onClick={()=>setActive(b.id)} aria-label={`Đọc ${b.title}`}><small>THƯ PHÒNG / {String(i+1).padStart(2,'0')}</small><strong>{b.title}</strong><span>TXT · {b.chapterCount} CHƯƠNG / PHẦN</span></button><div className="book-detail"><h3>{b.title}</h3><p>{b.author||'Tác giả chưa ghi'} · {b.charCount.toLocaleString('vi-VN')} ký tự</p><div><button className="text-button" onClick={()=>setActive(b.id)}>Mở sách →</button><button onClick={()=>setEdit({...b})}>Sửa</button><button onClick={async()=>{if(confirm(`Xóa “${b.title}” khỏi trình duyệt? Hãy sao lưu trước. File nguồn vẫn giữ nguyên.`)){await deleteBook(b.id);notify('Đã xóa bản lưu trong thư viện.');}}}>Xóa</button></div></div></article>)}</div>}
      {!!books?.length&&!filtered.length&&<p className="notice">Không có truyện khớp từ khóa.</p>}
    </section><div className="local-note"><span>◇</span><div><strong>Thư viện nằm trên thiết bị của bạn.</strong><p>Đổi máy hoặc xóa dữ liệu trình duyệt có thể làm mất bản lưu. Hãy sao lưu trước để mang câu chuyện theo mình.</p></div><button onClick={()=>setTab('backup')}>Sao lưu →</button></div>
    </>}
    {tab==='dictionary'&&<DictionaryPanel/>}{tab==='backup'&&<BackupPanel/>}{tab==='about'&&<section><div className="section-heading"><div><span className="eyebrow">PRIVATE NOVEL READER · 0.1</span><h1>Một thư phòng của riêng bạn</h1></div></div><div className="panel"><h2>Đọc và Convert trên thiết bị</h2><p>Nhập TXT và nạp từ điển không gửi nội dung truyện lên máy chủ. Một số giọng của trình duyệt dùng dịch vụ từ xa; ứng dụng sẽ hỏi trước khi dùng.</p><p>Bản này chưa đồng bộ cloud, chưa tích hợp AI, chưa có PWA mở lại hoàn toàn offline. Bản development chỉ dành localhost; bản online cần lớp đăng nhập Pages Functions.</p><button onClick={async()=>{const ok=await navigator.storage?.persist();notify(ok?'Trình duyệt đã cho phép lưu trữ bền vững. Vẫn cần sao lưu.':'Trình duyệt chưa cấp lưu trữ bền vững. Hãy sao lưu thường xuyên.');}}>Xin lưu trữ bền vững</button><button onClick={async()=>{await navigator.clipboard.writeText(location.origin);notify('Đã sao chép địa chỉ ứng dụng.');}}>Sao chép địa chỉ</button></div>
    {!!staged?.length&&<div className="panel"><h2>Bản nhập chưa hoàn tất ({staged.length})</h2><p>Những tác vụ này chưa được xác nhận vào thư viện. Có thể xóa phần tạm và nhập lại từ file gốc.</p>{staged.map(b=><div className="restore-row" key={b.id}><span>{b.title}</span><button onClick={()=>void deleteBook(b.id)}>Xóa bản tạm</button></div>)}</div>}
    </section>}
    <footer className="app-footer"><span>thư phòng</span><span>Đọc chậm lại. Đi xa hơn.</span></footer></main>
  </div>}
  {importing&&<ImportDialog onClose={()=>setImporting(false)} onDone={id=>{setImporting(false);setActive(id);notify('Đã thêm truyện vào thư viện.');}}/>}
  {edit&&<Modal title="Thông tin truyện" onClose={()=>setEdit(undefined)}><label>Tên truyện<input value={edit.title} onChange={e=>setEdit({...edit,title:e.target.value})}/></label><label>Tác giả<input value={edit.author} onChange={e=>setEdit({...edit,author:e.target.value})}/></label><footer><button className="primary" disabled={!edit.title.trim()} onClick={async()=>{await db.books.update(edit.id,{title:edit.title.trim(),author:edit.author,updatedAt:Date.now()});setEdit(undefined);}}>Lưu</button></footer></Modal>}
  {languageStatus&&<div className="worker-status" role="status">◌ {languageStatus}</div>}
  {toast&&<div className="toast" role="status"><span>{toast}</span><button onClick={()=>setToast('')} aria-label="Đóng thông báo">×</button></div>}
  </>;
}
