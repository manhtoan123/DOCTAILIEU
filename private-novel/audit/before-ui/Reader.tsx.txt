import {useCallback,useEffect,useMemo,useRef,useState,type CSSProperties} from 'react';
import {useLiveQuery} from 'dexie-react-hooks';
import {db} from '../db';
import {defaults,paragraphRanges,parseEntry,uid,type ReaderSettings,type Chapter} from '../domain';
import {convert,lookup} from '../language-client';
import type {Token} from '../language';
import {TikTokPlayer,tiktokVoices,voiceOptions,sentences,type SpeechState,type Sentence,type TikTokVoice,TtsProvider} from '../tts';
import {download} from '../backup';
import {Modal,errorText} from './common';
export default function Reader({bookId,onBack,notify}:{bookId:string;onBack:()=>void;notify:(s:string)=>void}){
  const book=useLiveQuery(()=>db.books.get(bookId),[bookId]);
  const list=useLiveQuery(async()=>{const items:Pick<Chapter,'id'|'title'|'order'>[]=[];await db.chapters.where('bookId').equals(bookId).each(c=>{items.push({id:c.id,title:c.title,order:c.order});});return items.sort((a,b)=>a.order-b.order);},[bookId]);
  const [chapterId,setChapterId]=useState('');const chapter=useLiveQuery(()=>chapterId?db.chapters.get(chapterId):undefined,[chapterId]);
  const names=useLiveQuery(()=>db.names.where('bookId').equals(bookId).toArray(),[bookId]);
  const [settings,setSettings]=useState<ReaderSettings>(defaults);const[controls,setControls]=useState(false);const[toc,setToc]=useState(false);const[tocFilter,setTocFilter]=useState('');const[tocPage,setTocPage]=useState(0);
  const[mode,setMode]=useState<'raw'|'convert'>('raw');const[tokens,setTokens]=useState<Token[]>([]);const[converting,setConverting]=useState(false);const[tempNames,setTempNames]=useState<{key:string;value:string}[]>([]);const[ttsTextOverride,setTtsTextOverride]=useState('');const[ttsEditor,setTtsEditor]=useState(false);const[ttsDraft,setTtsDraft]=useState<string[]>([]);
  const[edit,setEdit]=useState<string|null>(null);const editRef=useRef<HTMLTextAreaElement>(null);const[selected,setSelected]=useState<Token>();const[nameKey,setNameKey]=useState('');const[nameValue,setNameValue]=useState('');const[meaning,setMeaning]=useState({meanings:[] as string[],hanviet:''});const[namePanel,setNamePanel]=useState(false);
  const[autoNext,setAutoNext]=useState(true);const[speech,setSpeech]=useState<SpeechState>('idle');const[sentence,setSentence]=useState<Sentence>();
  const endedRef=useRef<()=>void>(()=>{});const continueRef=useRef(false);const playerRef=useRef<TikTokPlayer|null>(null);
  if(!playerRef.current)playerRef.current=new TikTokPlayer((state,s,msg)=>{setSpeech(state);setSentence(s);if(msg)notify(msg);},()=>endedRef.current());
  const progressTimer=useRef<ReturnType<typeof setTimeout>|undefined>(undefined);const position=useRef({paragraph:0,offset:0});const article=useRef<HTMLElement>(null);
  const save=useCallback(()=>{if(chapter)void db.progress.put({bookId,chapterId:chapter.id,paragraph:position.current.paragraph,offset:position.current.offset,revision:chapter.revision,updatedAt:Date.now()}).catch(e=>notify(errorText(e)));},[bookId,chapter,notify]);
  const settingsLoaded=useRef(false);
  useEffect(()=>{void db.settings.get('reader').then(row=>{if(row)setSettings({...defaults,...row.value as ReaderSettings});settingsLoaded.current=true;});},[]);
  useEffect(()=>{if(settingsLoaded.current)void db.settings.put({key:'reader',value:settings});},[settings]);
  useEffect(()=>{let alive=true;void db.progress.get(bookId).then(p=>{if(alive&&list?.length)setChapterId(old=>list.some(c=>c.id===old)?old:list.some(c=>c.id===p?.chapterId)?p!.chapterId:list[0].id);});return()=>{alive=false;};},[bookId,list]);
  useEffect(()=>()=>{playerRef.current?.stop();},[]);
  useEffect(()=>{let alive=true;setTokens([]);if(mode!=='convert'||!chapter){setConverting(false);return;}setConverting(true);void convert(chapter.text,[...(names||[]),...tempNames]).then(t=>{if(alive)setTokens(t);}).catch(e=>{if(alive){notify(errorText(e));setMode('raw');}}).finally(()=>{if(alive)setConverting(false);});return()=>{alive=false;};},[chapter?.id,chapter?.revision,mode,names,tempNames]);
  const renderedText=mode==='convert'?tokens.map(t=>t.targetText).join(''):chapter?.text||'';
  const speechText=ttsTextOverride||renderedText;
  const paragraphs=useMemo(()=>paragraphRanges(renderedText),[renderedText]);
  const blocks=useMemo(()=>{
    if(mode==='raw')return [];
    const rows:{token:Token;text:string}[][]=[[]];for(const token of tokens){const pieces=token.targetText.split('\n');pieces.forEach((text,i)=>{if(text)rows[rows.length-1].push({token,text});if(i<pieces.length-1)rows.push([]);});}return rows;
  },[tokens,mode]);
  useEffect(()=>{if(!chapter||converting||!paragraphs.length)return;let alive=true;let ready=false;let timeout:ReturnType<typeof setTimeout>;let observer:IntersectionObserver|undefined;
    void db.progress.get(bookId).then(p=>{if(!alive)return;const target=p?.chapterId===chapter.id?Math.min(p.paragraph,paragraphs.length-1):0;position.current={paragraph:target,offset:0};timeout=setTimeout(()=>{if(!alive)return;article.current?.querySelector(`[data-paragraph="${target}"]`)?.scrollIntoView({block:'start'});ready=true;},80);
      observer=new IntersectionObserver(entries=>{if(!ready)return;const visible=entries.filter(e=>e.isIntersecting).sort((a,b)=>a.boundingClientRect.top-b.boundingClientRect.top);if(visible[0]){position.current.paragraph=Number((visible[0].target as HTMLElement).dataset.paragraph);clearTimeout(progressTimer.current);progressTimer.current=setTimeout(save,600);}}, {rootMargin:'-15% 0px -60% 0px'});
      article.current?.querySelectorAll('[data-paragraph]').forEach(p=>observer!.observe(p));
    });const visibility=()=>{if(ready)save();};document.addEventListener('visibilitychange',visibility);return()=>{alive=false;clearTimeout(timeout);clearTimeout(progressTimer.current);observer?.disconnect();document.removeEventListener('visibilitychange',visibility);if(ready)save();};
  },[chapter?.id,chapter?.revision,converting,renderedText]);
  const currentIndex=list?.findIndex(c=>c.id===chapterId)??0;
  function changeChapter(id:string,keepPlaying=false){save();playerRef.current?.stop();continueRef.current=keepPlaying;setChapterId(id);setToc(false);}
  function openTtsEditor(){setTtsDraft(sentences(speechText).map(item=>item.text));setTtsEditor(true);}
  function saveTtsDraft(){const text=ttsDraft.map(item=>item.trim()).filter(Boolean).join('\n');setTtsTextOverride(text);setTtsEditor(false);playerRef.current?.stop();notify('Đã áp dụng văn bản nghe cho phiên này.');}
  function play(){if(!speechText.trim())return;const option=voiceOptions.find(v=>v.id===settings.voiceId);if(!option?.available||option.provider!=='tiktok'){notify('Gateway của giọng này chưa được cấu hình; hiện TikTok TTS là giọng được hỗ trợ.');return;}playerRef.current!.start(speechText,option as TikTokVoice,{rate:settings.ttsRate,pitch:settings.ttsPitch,flex:settings.ttsFlex});}
  endedRef.current=()=>{if(autoNext&&list?.[currentIndex+1])changeChapter(list[currentIndex+1].id,true);};
  useEffect(()=>{setTtsTextOverride('');},[chapter?.id,mode]);
  useEffect(()=>{if(continueRef.current&&chapter&&!converting&&speechText){continueRef.current=false;const option=voiceOptions.find(v=>v.id===settings.voiceId)||tiktokVoices[0];if(option.provider==='tiktok')playerRef.current!.start(speechText,option as TikTokVoice,{rate:settings.ttsRate,pitch:settings.ttsPitch,flex:settings.ttsFlex});}},[chapter?.id,speechText,converting,settings.voiceId,settings.ttsRate,settings.ttsPitch,settings.ttsFlex]);
  async function selectToken(token:Token){setSelected(token);setNameKey(token.source);setNameValue(token.targetText.trim());try{setMeaning(await lookup(token.source));}catch(e){notify(errorText(e));}}
  async function saveName(permanent:boolean){if(!nameKey.trim()||!nameValue.trim())return;if(nameKey.length>256||nameValue.length>4096){notify('Tên hoặc nghĩa quá dài.');return;}if(permanent){const old=await db.names.where('[bookId+key]').equals([bookId,nameKey.trim()]).first();await db.names.put({id:old?.id||uid(),bookId,key:nameKey.trim(),value:nameValue.trim()});}else setTempNames(items=>[...items.filter(n=>n.key!==nameKey),{key:nameKey.trim(),value:nameValue.trim()}]);setSelected(undefined);notify(permanent?'Đã lưu Name cho truyện này.':'Đã sửa tạm trong phiên đọc.');}
  async function saveEdit(splitAt?:number){if(!chapter||edit===null)return;if(edit.length>128000){notify('Mỗi phần tối đa 128.000 ký tự. Hãy chia thành nhiều phần.');return;}const delta=edit.length-chapter.text.length;
    await db.transaction('rw',[db.chapters,db.books],async()=>{if(splitAt&&splitAt>0&&splitAt<edit.length){const next=list?.[currentIndex+1];const order=next?(chapter.order+next.order)/2:chapter.order+1;await db.chapters.add({...chapter,id:uid(),order,title:chapter.title+' · tiếp',text:edit.slice(splitAt),revision:1});await db.chapters.update(chapter.id,{text:edit.slice(0,splitAt),revision:chapter.revision+1});await db.books.update(bookId,{chapterCount:(book?.chapterCount||0)+1,charCount:(book?.charCount||0)+delta,updatedAt:Date.now()});}else{await db.chapters.update(chapter.id,{text:edit,revision:chapter.revision+1});await db.books.update(bookId,{charCount:(book?.charCount||0)+delta,updatedAt:Date.now()});}});setEdit(null);playerRef.current?.stop();}
  const filtered=list?.filter(c=>c.title.toLocaleLowerCase().includes(tocFilter.toLocaleLowerCase()))||[];
  return <div className={`reader theme-${settings.theme}`} style={{'--tts-highlight':settings.highlightColor} as CSSProperties}>
    <header className="reader-header"><button onClick={()=>{save();onBack();}}>← Tủ sách</button><div><small>{book?.title}</small><strong>{chapter?.title||'Đang mở…'}</strong></div><button onClick={()=>setToc(true)}>☷ Mục lục</button><button onClick={()=>setControls(v=>!v)}>Aa <span className="desktop-only">Hiển thị</span></button></header>
    {controls&&<section className="reader-controls"><label>Nền<select value={settings.theme} onChange={e=>setSettings(s=>({...s,theme:e.target.value as ReaderSettings['theme']}))}><option value="light">Sáng</option><option value="sepia">Sepia</option><option value="dark">Tối</option><option value="gray">Xám đêm</option></select></label><label>Font<select value={settings.font} onChange={e=>setSettings(s=>({...s,font:e.target.value as 'serif'|'sans'}))}><option value="serif">Serif · đọc sách</option><option value="sans">Sans · rõ nét</option></select></label><label>Cỡ chữ {settings.fontSize}<input type="range" min="15" max="34" value={settings.fontSize} onChange={e=>setSettings(s=>({...s,fontSize:+e.target.value}))}/></label><label>Giãn dòng {settings.lineHeight}<input type="range" min="1.3" max="2.6" step="0.1" value={settings.lineHeight} onChange={e=>setSettings(s=>({...s,lineHeight:+e.target.value}))}/></label><label>Chiều rộng<input type="range" min="500" max="1100" step="20" value={settings.width} onChange={e=>setSettings(s=>({...s,width:+e.target.value}))}/></label></section>}
    <div className="reader-actions"><div className="segmented"><button className={mode==='raw'?'active':''} onClick={()=>{playerRef.current?.stop();setMode('raw');}}>Bản gốc</button><button className={mode==='convert'?'active':''} onClick={()=>{playerRef.current?.stop();setMode('convert');}}>Convert</button></div><button onClick={()=>setNamePanel(true)}>Name List ({names?.length||0})</button><button onClick={()=>setEdit(chapter?.text||'')}>Sửa chương</button><button onClick={()=>{if(chapter)download(new Blob([chapter.text],{type:'text/plain;charset=utf-8'}),`${chapter.title.replace(/[<>:"/\\|?*]/g,'_')}.txt`);}}>Xuất TXT</button></div>
    <article ref={article} className="prose" style={{maxWidth:settings.width,fontSize:settings.fontSize,lineHeight:settings.lineHeight,fontFamily:settings.font==='serif'?'Georgia, "Times New Roman", serif':'system-ui, sans-serif'}}>
      <div className="chapter-kicker">CHƯƠNG / PHẦN {Math.max(0,currentIndex)+1} TRÊN {list?.length||0}</div><h1>{chapter?.title}</h1>
      {converting?<p className="notice">Đang Convert trong Worker…</p>:paragraphs.map((p,i)=><p key={`${chapterId}-${mode}-${i}`} data-paragraph={i} className={sentence&&sentence.start<p.start+p.text.length&&sentence.end>p.start?'speaking':''}>{mode==='convert'?(blocks[i]||[]).map((x,j)=><span key={j} className={x.token.type!=='original'?'translated-token':''} onClick={()=>void selectToken(x.token)}>{x.text}</span>):p.text}{!p.text.trim()&&'\u00a0'}</p>)}
      <nav className="chapter-navigation"><button disabled={currentIndex<=0} onClick={()=>changeChapter(list![currentIndex-1].id)}>← Chương trước</button><button disabled={!list?.[currentIndex+1]} onClick={()=>changeChapter(list![currentIndex+1].id)}>Chương sau →</button></nav>
    </article>
    <div className="tts-bar"><button className="primary" disabled={converting||!chapter} onClick={()=>speech==='playing'?playerRef.current?.pause():speech==='paused'?playerRef.current?.resume():play()}>{speech==='playing'?'Ⅱ Tạm dừng':speech==='paused'?'▶ Tiếp tục':speech==='loading'?'Đang tải TikTok…':'▶ Nghe TikTok'}</button><button aria-label="Dừng đọc" onClick={()=>playerRef.current?.stop()}>■</button><button onClick={openTtsEditor}>Câu ({sentences(speechText).length})</button><label>Giọng<select aria-label="Giọng đọc" value={settings.voiceId} onChange={e=>{playerRef.current?.stop();setSettings(s=>({...s,voiceId:e.target.value}));}}>{(['bing','google-cloud','google-translate','tiktok'] as TtsProvider[]).map(provider=><optgroup key={provider} label={provider==='bing'?'Dịch vụ giọng nói Bing':provider==='google-cloud'?'Dịch vụ giọng nói Google Cloud':provider==='google-translate'?'Giọng Chị Google':'Giọng đọc của TikTok'}>{voiceOptions.filter(v=>v.provider===provider).map(v=><option key={v.id} value={v.id}>{v.name}{v.available?'':' · chưa cấu hình'}</option>)}</optgroup>)}</select></label><label>Tốc độ<input aria-label="Tốc độ đọc" type="range" min="0.5" max="5" step="0.1" value={settings.ttsRate} onChange={e=>setSettings(s=>({...s,ttsRate:+e.target.value}))}/></label><output>{settings.ttsRate}</output><label>Cao độ<input aria-label="Độ cao giọng" type="range" min="0.5" max="2" step="0.1" value={settings.ttsPitch} onChange={e=>setSettings(s=>({...s,ttsPitch:+e.target.value}))}/></label><output>{settings.ttsPitch}</output><label>Ngữ cảnh<input aria-label="Điều chỉnh theo ngữ cảnh" type="range" min="0" max="2" step="0.1" value={settings.ttsFlex} onChange={e=>setSettings(s=>({...s,ttsFlex:+e.target.value}))}/></label><output>{settings.ttsFlex}</output><label>Màu highlight<input aria-label="Màu làm nổi bật văn bản" type="color" value={settings.highlightColor} onChange={e=>setSettings(s=>({...s,highlightColor:e.target.value}))}/></label><button onClick={()=>void db.settings.put({key:'reader',value:settings})}>Lưu cài đặt</button><label className="check"><input type="checkbox" checked={autoNext} onChange={e=>setAutoNext(e.target.checked)}/>Qua chương</label><small>TikTok TTS · tiếng Việt</small></div>
    {toc&&<Modal title="Mục lục" onClose={()=>setToc(false)}><input aria-label="Tìm chương" placeholder="Tìm chương…" value={tocFilter} onChange={e=>{setTocFilter(e.target.value);setTocPage(0);}}/><div className="toc">{filtered.slice(tocPage*60,tocPage*60+60).map(c=><button className={c.id===chapterId?'current':''} key={c.id} onClick={()=>changeChapter(c.id)}>{c.title}</button>)}</div><div className="pager"><button disabled={!tocPage} onClick={()=>setTocPage(p=>p-1)}>Trước</button><span>{tocPage+1}</span><button disabled={(tocPage+1)*60>=filtered.length} onClick={()=>setTocPage(p=>p+1)}>Tiếp</button></div></Modal>}
    {edit!==null&&<Modal title="Sửa nội dung chương" onClose={()=>setEdit(null)} wide><p className="muted">Sửa bản lưu trong ứng dụng. File nguồn giữ nguyên. Đặt con trỏ để tách chương tại vị trí đó.</p><textarea ref={editRef} aria-label="Nội dung chương" rows={18} value={edit} onChange={e=>setEdit(e.target.value)}/><footer><button onClick={()=>void saveEdit(editRef.current?.selectionStart)}>Tách tại con trỏ</button><button className="primary" onClick={()=>void saveEdit()}>Lưu thay đổi</button></footer></Modal>}
    {selected&&<Modal title="Chỉnh Name / nghĩa" onClose={()=>setSelected(undefined)}><label>Chữ gốc<input value={nameKey} onChange={e=>setNameKey(e.target.value)}/></label><div className="pager"><button disabled={!chapter||selected.sourceStart<=0} onClick={()=>{if(chapter)void selectToken({...selected,sourceStart:Math.max(0,selected.sourceStart-1),source:chapter.text.slice(Math.max(0,selected.sourceStart-1),selected.sourceEnd)});}}>← Mở rộng trái</button><button disabled={!chapter||selected.sourceEnd>=chapter.text.length} onClick={()=>{if(chapter)void selectToken({...selected,sourceEnd:selected.sourceEnd+1,source:chapter.text.slice(selected.sourceStart,selected.sourceEnd+1)});}}>Mở rộng phải →</button></div><p>Hán-Việt: {meaning.hanviet||'Chưa có phiên âm'}</p><div className="choices">{meaning.meanings.slice(0,100).map((m,i)=><button key={i} onClick={()=>setNameValue(m)}>{m}</button>)}</div><label>Nghĩa / tên Việt<input value={nameValue} onChange={e=>setNameValue(e.target.value)}/></label><div className="choices"><button onClick={()=>setNameValue(v=>v.toLocaleUpperCase('vi'))}>CHỮ HOA</button><button onClick={()=>setNameValue(v=>v.toLocaleLowerCase('vi'))}>chữ thường</button></div><footer><button onClick={()=>void saveName(false)}>Sửa tạm</button><button className="primary" onClick={()=>void saveName(true)}>Lưu Name cho truyện</button></footer></Modal>}
    {namePanel&&<Modal title="Name List riêng của truyện" onClose={()=>setNamePanel(false)}><p className="muted">Mỗi dòng chữ Trung=tên Việt. Import cập nhật key trùng trong truyện này.</p><input type="file" accept=".txt" aria-label="Nhập Name List" onChange={async e=>{const f=e.target.files?.[0];if(!f)return;try{if(f.size>2*1048576)throw new Error('Name List tối đa 2 MB.');const entries=(await f.text()).split(/\r?\n/).map(parseEntry).filter((e):e is [string,string]=>!!e);await db.transaction('rw',db.names,async()=>{for(const[key,value]of entries){if(key.length>256||value.length>4096)continue;const old=await db.names.where('[bookId+key]').equals([bookId,key]).first();await db.names.put({id:old?.id||uid(),bookId,key,value});}});notify(`Đã xử lý ${entries.length} dòng Name.`);}catch(err){notify(errorText(err));}}}/><button onClick={()=>download(new Blob([(names||[]).map(n=>`${n.key}=${n.value}`).join('\n')]),'NameList.txt')}>Xuất Name List</button><div className="name-list">{names?.slice(0,200).map(n=><div key={n.id}><span>{n.key} = {n.value}</span><button aria-label={`Xóa Name ${n.key}`} onClick={()=>void db.names.delete(n.id)}>×</button></div>)}</div>{(names?.length||0)>200&&<p className="muted">Hiển thị 200 mục đầu. Xuất TXT để chỉnh toàn bộ danh sách.</p>}</Modal>}
  </div>;
}
